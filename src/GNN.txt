import torch
import torch.nn as nn
import torch.nn.functional as F
from torch_geometric.data import Data
from torch_geometric.nn import GCNConv
from torch.optim import Adam

# Define a simple Graph Neural Network model
class GNN(nn.Module):
    def __init__(self, num_nodes, embedding_dim):
        super(GNN, self).__init__()
        self.conv1 = GCNConv(embedding_dim, embedding_dim)
        self.conv2 = GCNConv(embedding_dim, embedding_dim)
    
    def forward(self, x, edge_index):
        x = F.relu(self.conv1(x, edge_index))
        x = self.conv2(x, edge_index)
        return x

# Process text data to extract entities and relationships with context
text_data = [
    ("Tom likes apples in the morning", "like"),
    ("Alice eats oranges in the evening", "eat"),
    ("Bob dislikes bananas in the afternoon", "dislike")
]

# Create a dictionary mapping entities to unique IDs
entity_to_id = {entity: idx for idx, (entity, _) in enumerate(set([item for sublist in [sentence.split() for sentence, _ in text_data] for item in sublist]))}

# Convert text data into a graph structure with context
edge_index = []
for sentence, relation in text_data:
    tokens = sentence.split()
    context = " ".join(tokens[-3:])  # Consider last three words as context
    head, relation, tail = tokens[:-3]
    edge_index.append([entity_to_id[head], entity_to_id[tail], entity_to_id[context]])

edge_index = torch.tensor(edge_index, dtype=torch.long).t()

num_nodes = len(entity_to_id)
node_features = torch.eye(num_nodes)  # One-hot encoded node features

# Define the model, optimizer, and loss function
model = GNN(num_nodes, embedding_dim=64)
optimizer = Adam(model.parameters(), lr=0.01)
criterion = nn.BCELoss()

# Training loop
num_epochs = 50
for epoch in range(num_epochs):
    model.train()
    optimizer.zero_grad()
    
    # Forward pass
    output = model(node_features, edge_index)
    
    # Compute reconstruction loss
    loss = criterion(output, node_features)
    
    # Backward pass and optimization step
    loss.backward()
    optimizer.step()
    
    print(f"Epoch {epoch+1}, Loss: {loss.item()}")

# Inference for knowledge graph completion
model.eval()
completed_graph = model(node_features, edge_index)

# Extract predicted edges from the completed graph
predicted_edges = torch.round(torch.sigmoid(torch.matmul(completed_graph, completed_graph.t())))

print("Predicted edges:")
print(predicted_edges)
